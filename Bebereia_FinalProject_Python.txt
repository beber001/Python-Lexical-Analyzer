def convert(tokens):
  """ convert lexer output to parser input 
      e.g. ['<keyword,int>',...] -> [('keyword','int'),...]
  """
  new_tokens = []
  for token in tokens:
    t = token[1:-1]  # strip off < and >
    parts = t.split(',')
    new_tokens.append((parts[0], parts[1]))
  return new_tokens


def CutOneLineTokens(s):
  tokens = []
  patt_kw = r'(if +|else +|int +|float +|print+ |string+)'
  patt_sep = r'[\(\)\:;]'
  patt_ops = r'[+=>\*]'
  patt_ident = r'([A-z]+\d*)'
  patt_int = r'(\d+)'
  patt_float = r'(\d+\.\d+)'
  patt_string = r'\"[^\"]*\"'
  
  while True:
    s = s.strip()
    if len(s) == 0:
      break  # done parsing
    # keyword
    m = re.match(patt_kw, s)
    if m:
      kw = m.group().strip()
      tokens.append('<keyword,' + kw + '>')
      s = s[m.end():]
      continue
    # seperator
    m = re.match(patt_sep, s)
    if m:
      sr = m.group()
      tokens.append('<seperator,' + sr + '>')
      s = s[m.end():]
      continue
    # operator
    m = re.match(patt_ops, s)
    if m:
      op = m.group()
      tokens.append('<operator,' + op + '>')
      s = s[m.end():]
      continue
    # identifier
    m = re.match(patt_ident, s)
    if m:
      iden = m.group()
      tokens.append('<identifier,' + iden + '>')
      s = s[m.end():]
      continue

    # float
    m = re.match(patt_float, s)
    if m:
      sewer = m.group()
      tokens.append('<float,' + sewer + '>')
      s = s[m.end():]
      continue
    # integer
    m = re.match(patt_int, s)
    if m:
      aaa = m.group()
      tokens.append('<integer,' + aaa + '>')
      s = s[m.end():]
      continue

    # string
    print(s)
    m = re.match(patt_string, s)
    if m:
      cat = m.group()
      tokens.append('<separator,">')
      tokens.append('<string,' + cat[1:-1] + '>')
      tokens.append('<separator,">')
      s = s[m.end():]
      continue

  return tokens

  def inorderTraversal(self, root):

  def clear_tree(self):
    for item in self.treeview.get_children():
      self.treeview.delete(item)

  def make_tree(self,otherTree):
    self.treeview.insert('', 0, 'item0', text ='Expression')
    index = 1
    parent_list = ['item0']  
    while otherTree:
      index += 1
      item = otherTree.pop(0)
      name = 'item' + str(index)
      # down a level
      if item == '(': 
        new_parent_text = otherTree.pop(0)
        self.treeview.insert(parent_list[-1],index, name, text=new_parent_text)
        parent_list.append(name)
        continue
      # up a level
      if item == ')':
        parent_list.pop()
        continue  
      # add item
      self.treeview.insert(parent_list[-1], index, name, text=item)
